cmake_minimum_required(VERSION 3.19)
project(wire_libcxx_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  set(SYSIO_WASM_RUNTIMES native-module)
  include(../../native-libcxx.cmake)
endif()
include(../../check-libcxx.cmake)
find_package(Threads REQUIRED)
add_executable(libcxx_runtime_test ../../../tests/libcxx_runtime_test.cpp ../../../tests/libcxx_c_runtime.c)
target_link_libraries(libcxx_runtime_test PRIVATE Threads::Threads)
enable_testing()
add_test(NAME libcxx_runtime_test COMMAND libcxx_runtime_test)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  add_library(native_libcxx_module MODULE native_module.cpp)
  add_executable(native_libcxx_host native_host.cpp)
  target_link_libraries(native_libcxx_host PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME native_libcxx_test COMMAND native_libcxx_host $<TARGET_FILE:native_libcxx_module>)
endif()
